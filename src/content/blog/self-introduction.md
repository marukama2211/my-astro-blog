```python
import numpy as np

a = np.arange(1, 10)
r = np.random.default_rng(0).integers(1, 10, 9)

A = a.reshape(3, 3)
B = r.reshape(3, 3)
print(A + B)
print(A * B)
print(A @ B)
print(A.T)

v = np.array([1, 0, -1])
print(A @ v)

M = np.array([[2.0, 1.0], [1.0, 3.0]])
y = np.array([3.0, 5.0])
x = np.linalg.solve(M, y)
print(x)
print(M @ x)

print(np.linalg.inv(M))
print(np.linalg.det(M))
vals, vecs = np.linalg.eig(M)
print(vals)
```
